// Routes de gestion des demandes (formulaires soumis par les agents). // Toutes ces routes nécessitent une session active (middleware requireAuth) : // un agent ne peut voir ou créer que ses propres demandes. const express = require('express'); const crypto = require('crypto'); const { body, validationResult } = require('express-validator'); const { pool } = require('../db/pool'); const { requireAuth } = require('../middleware/requireAuth'); const router = express.Router(); // Préfixe de référence par formulaire, repris des données déjà définies // côté front (categoryForms dans public/index.html) — voir la constante // REQUEST_REF_PREFIXES dans index.html pour la liste complète si de // nouveaux formulaires sont ajoutés un jour. const REF_PREFIXES = { arrivee: 'ARR', demenagement: 'DEM', depart: 'DEP', signaletique: 'SIG', cartons: 'CAE', dossier_srh: 'SRH', courantes: 'FCO', sanitaires: 'FSA', specifiques: 'FSP', tampons: 'TAM', menage_general: 'MEN', ascenseur: 'ASC', intervention: 'ITE', acces_exceptionnel: 'AEX', perte_badge: 'PBA', cle_bureau: 'CLE', acces_prestataire: 'APR', modif_droits: 'MDB', livraison: 'LIV', signaler_envoi: 'SEC', manutention: 'MAN', livraison_quai: 'LEQ', mobilier_standard: 'MST', equip_specifiques: 'ESB', reservation: 'VEH', }; function generateReference(formKey) { const prefix = REF_PREFIXES[formKey] || 'REQ'; const year = new Date().getFullYear(); const random = Math.floor(1000 + Math.random() * 9000); return `${prefix}-${year}-${random}`; } // Génère une référence garantie unique en base, en retentant en cas de // collision improbable (l'unicité est aussi imposée par une contrainte // UNIQUE en base — voir schema.sql — donc même une collision non détectée // ici serait rattrapée par une erreur d'insertion explicite). async function generateUniqueReference(client, formKey) { for (let attempt = 0; attempt < 5; attempt += 1) { const ref = generateReference(formKey); const existing = await client.query('SELECT 1 FROM requests WHERE reference = $1', [ref]); if (existing.rows.length === 0) return ref; } // Cas extrêmement improbable après 5 tentatives : on retombe sur un // suffixe totalement aléatoire pour garantir l'unicité. const prefix = REF_PREFIXES[formKey] || 'REQ'; return `${prefix}-${new Date().getFullYear()}-${crypto.randomBytes(3).toString('hex').toUpperCase()}`; } // --------------------------------------------------------------- // POST /api/requests — soumettre un nouveau formulaire // --------------------------------------------------------------- router.post( '/', requireAuth, [ body('categoryKey').trim().notEmpty().withMessage('La catégorie est requise.'), body('formKey').trim().notEmpty().withMessage('Le formulaire est requis.'), body('title').trim().notEmpty().withMessage('Le titre du formulaire est requis.'), body('data').isObject().withMessage('Les données du formulaire sont invalides.'), body('status') .optional() .isIn(['brouillon', 'en_cours']) .withMessage("Le statut à la création doit être 'brouillon' ou 'en_cours'."), ], async (req, res) => { const errors = validationResult(req); if (!errors.isEmpty()) { return res.status(400).json({ error: errors.array()[0].msg }); } const { categoryKey, formKey, title, data } = req.body; const status = req.body.status || 'en_cours'; const client = await pool.connect(); try { const reference = await generateUniqueReference(client, formKey); const result = await client.query( `INSERT INTO requests (user_id, reference, category_key, form_key, title, status, data) VALUES ($1, $2, $3, $4, $5, $6, $7) RETURNING id, reference, category_key, form_key, title, status, data, created_at, updated_at`, [req.session.userId, reference, categoryKey, formKey, title, status, JSON.stringify(data)] ); const created = result.rows[0]; await client.query( `INSERT INTO request_history (request_id, previous_status, new_status, note) VALUES ($1, NULL, $2, 'Création de la demande')`, [created.id, status] ); return res.status(201).json({ request: created }); } catch (err) { console.error('[requests/create] Erreur :', err); return res.status(500).json({ error: 'Une erreur est survenue, veuillez réessayer.' }); } finally { client.release(); } } ); // --------------------------------------------------------------- // GET /api/requests — lister les demandes de l'agent connecté // --------------------------------------------------------------- router.get('/', requireAuth, async (req, res) => { try { const result = await pool.query( `SELECT id, reference, category_key, form_key, title, status, data, created_at, updated_at FROM requests WHERE user_id = $1 ORDER BY updated_at DESC`, [req.session.userId] ); return res.json({ requests: result.rows }); } catch (err) { console.error('[requests/list] Erreur :', err); return res.status(500).json({ error: 'Une erreur est survenue, veuillez réessayer.' }); } }); // --------------------------------------------------------------- // GET /api/requests/:id — détail d'une demande (avec ses données complètes // et son historique de statuts) // --------------------------------------------------------------- router.get('/:id', requireAuth, async (req, res) => { const { id } = req.params; try { const requestResult = await pool.query( `SELECT id, reference, category_key, form_key, title, status, data, created_at, updated_at FROM requests WHERE id = $1 AND user_id = $2`, [id, req.session.userId] ); if (requestResult.rows.length === 0) { // 404 générique, qu'il s'agisse d'un id inexistant ou d'une demande // appartenant à un autre agent — on ne révèle pas la différence. return res.status(404).json({ error: 'Demande introuvable.' }); } const historyResult = await pool.query( `SELECT previous_status, new_status, note, created_at FROM request_history WHERE request_id = $1 ORDER BY created_at ASC`, [id] ); return res.json({ request: requestResult.rows[0], history: historyResult.rows }); } catch (err) { console.error('[requests/detail] Erreur :', err); return res.status(500).json({ error: 'Une erreur est survenue, veuillez réessayer.' }); } }); // --------------------------------------------------------------- // PATCH /api/requests/:id — met à jour un brouillon existant (sauvegarde // progressive pendant le remplissage d'un formulaire). Volontairement // limité aux demandes actuellement au statut 'brouillon' : une fois la // demande envoyée ('en_cours' ou au-delà), elle ne doit plus pouvoir être // modifiée par cette route — toute évolution ultérieure du statut passera // par un mécanisme dédié au traitement, distinct de la simple sauvegarde // d'un formulaire en cours de saisie. // --------------------------------------------------------------- router.patch( '/:id', requireAuth, [ body('data').isObject().withMessage('Les données du formulaire sont invalides.'), body('status') .optional() .isIn(['brouillon', 'en_cours']) .withMessage("Le statut doit être 'brouillon' ou 'en_cours'."), ], async (req, res) => { const errors = validationResult(req); if (!errors.isEmpty()) { return res.status(400).json({ error: errors.array()[0].msg }); } const { id } = req.params; const { data } = req.body; const nextStatus = req.body.status || 'brouillon'; const client = await pool.connect(); try { const existing = await client.query( `SELECT id, status FROM requests WHERE id = $1 AND user_id = $2`, [id, req.session.userId] ); if (existing.rows.length === 0) { return res.status(404).json({ error: 'Demande introuvable.' }); } if (existing.rows[0].status !== 'brouillon') { return res.status(409).json({ error: 'Cette demande a déjà été envoyée et ne peut plus être modifiée comme un brouillon.', }); } const result = await client.query( `UPDATE requests SET data = $1, status = $2, updated_at = now() WHERE id = $3 RETURNING id, reference, category_key, form_key, title, status, data, created_at, updated_at`, [JSON.stringify(data), nextStatus, id] ); if (nextStatus !== existing.rows[0].status) { await client.query( `INSERT INTO request_history (request_id, previous_status, new_status, note) VALUES ($1, $2, $3, $4)`, [id, existing.rows[0].status, nextStatus, nextStatus === 'en_cours' ? 'Envoi de la demande' : 'Mise à jour du brouillon'] ); } return res.json({ request: result.rows[0] }); } catch (err) { console.error('[requests/update] Erreur :', err); return res.status(500).json({ error: 'Une erreur est survenue, veuillez réessayer.' }); } finally { client.release(); } } ); module.exports = router;
